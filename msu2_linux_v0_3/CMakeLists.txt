cmake_minimum_required(VERSION 3.16)
project(msu2_mini_linux VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(msu2_core src/protocol.cpp src/delta.cpp src/image.cpp src/session.cpp src/transport.cpp src/capture.cpp src/priority.cpp)
target_include_directories(msu2_core PUBLIC include)
target_link_libraries(msu2_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(msu2_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(msu2_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(msu2_core PUBLIC user32 gdi32)
endif()
add_executable(msu2 src/main.cpp)
target_link_libraries(msu2 PRIVATE msu2_core)
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_core.cpp")
  add_executable(msu2_tests tests/test_core.cpp)
  target_link_libraries(msu2_tests PRIVATE msu2_core)
  add_test(NAME core COMMAND msu2_tests)
  add_executable(msu2_delta_tests tests/test_delta.cpp)
  target_link_libraries(msu2_delta_tests PRIVATE msu2_core)
  add_test(NAME delta COMMAND msu2_delta_tests)
  add_executable(msu2_priority_tests tests/test_priority.cpp)
  target_link_libraries(msu2_priority_tests PRIVATE msu2_core)
  add_test(NAME priority COMMAND msu2_priority_tests)
endif()
